cmake_minimum_required(VERSION 3.25)
project(Project8SwitchProbes VERSION 0.1.0 LANGUAGES C CXX)
if(NOT NINTENDO_SWITCH OR NOT CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
    message(FATAL_ERROR "Use tools/cross/switch-devkita64.cmake")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(memory_probe memory_probe/main.cpp)
target_compile_definitions(memory_probe PRIVATE REX_PLATFORM_SWITCH=1)
target_compile_options(memory_probe PRIVATE -Wall -Wextra -Werror)
nx_create_nro(memory_probe)
add_executable(toolchain_probe toolchain_probe/main.cpp)
target_compile_options(toolchain_probe PRIVATE -Wall -Wextra -Werror -fno-strict-aliasing)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(toolchain_probe PRIVATE -ffp-model=strict)
endif()
nx_create_nro(toolchain_probe)
option(SWITCH_BUILD_VULKAN_PROBE "Build external nxvk Vulkan diagnostic" OFF)
if(SWITCH_BUILD_VULKAN_PROBE)
    add_subdirectory(vulkan_probe)
endif()
